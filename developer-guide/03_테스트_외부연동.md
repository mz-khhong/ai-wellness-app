```java
@ExtendWith(MockitoExtension.class)
class AdminServiceTest {
    
    @Mock
    private AdminRepositoryPort adminRepositoryPort;
    
    @Mock
    private PasswordEncoder passwordEncoder;
    
    @InjectMocks
    private AdminService adminService;
    
    private Admin mockAdmin;
    
    @BeforeEach
    void setUp() {
        // MockData를 사용하여 테스트용 객체 생성
        mockAdmin = MockData.createAdmin();
        mockAdmin = Admin.builder()
                .id(1L)
                .facilityGroupId(1L)
                .uuid("test-uuid-123")
                .email(mockAdmin.getEmail())
                .name(mockAdmin.getName())
                .password(mockAdmin.getPassword())
                .status(mockAdmin.getStatus())
                .createdAt(mockAdmin.getCreatedAt())
                .updatedAt(mockAdmin.getUpdatedAt())
                .build();
    }
    
    @Test
    void getAdmin_존재하는_관리자_조회_성공() {
        // Given
        Long id = 1L;
        when(adminRepositoryPort.findById(id))
                .thenReturn(Optional.of(mockAdmin));
        
        // When
        AdminResponse result = adminService.getAdmin(id);
        
        // Then
        assertThat(result).isNotNull();
        assertThat(result.getId()).isEqualTo(id);
        assertThat(result.getEmail()).isEqualTo(mockAdmin.getEmail());
    }
    
    @Test
    void getAdmin_존재하지_않는_관리자_조회_실패() {
        // Given
        Long id = 999L;
        when(adminRepositoryPort.findById(id))
                .thenReturn(Optional.empty());
        
        // When & Then
        assertThatThrownBy(() -> adminService.getAdmin(id))
                .isInstanceOf(AdminBusinessException.class)
                .hasMessageContaining("admin.not.found");
    }
}
```

## 3. 외부/내부 서비스 연동 (Feign Client)

### 3.1. Feign Client 개요

- **내부 서비스 연동**: admin-server, manager-server, customer-server 간 호출
- **외부 서비스 연동**: 결제, 이메일, SMS 등 외부 API 호출

### 3.2. 폴더 구조

```
{server}-server/src/main/java/com/aiwellness/{server}/
├── adapter/
│   └── infrastructure/
│       └── client/                    # Feign Client
│           ├── admin/                 # Admin Server 호출
│           │   └── AdminServiceClient.java
│           ├── manager/               # Manager Server 호출
│           │   └── ManagerServiceClient.java
│           └── customer/              # Customer Server 호출
│               └── CustomerServiceClient.java
│
└── domain/
    └── port/
        └── external/                   # 외부 서비스 Port
            ├── AdminServicePort.java
            ├── ManagerServicePort.java
            └── CustomerServicePort.java
```

### 3.3. 구현 방법

#### 1단계: Domain Port 정의
```java

 * Admin 서버 호출을 위한 Port 인터페이스
 */

    AdminResponse createAdmin(AdminCreateRequest request);

```

#### 2단계: Feign Client 구현
```java
package com.aiwellness.{server}.adapter.infrastructure.client.admin;

import com.aiwellness.{server}.domain.port.external.AdminServicePort;
import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;

/**
 * Admin 서버 호출을 위한 Feign Client
 */
@FeignClient(
    name = "admin-service",
    url = "${services.admin.url}",
    path = "/api"
)
public interface AdminServiceClient extends AdminServicePort {
    
    @GetMapping("/admins/{id}")
    @Override
    AdminResponse getAdmin(@PathVariable Long id);
    
    @PostMapping("/admins")
    @Override
    AdminResponse createAdmin(@RequestBody AdminCreateRequest request);
}
```

#### 3단계: Adapter 구현 (외부 호출 로그 포함)
```java
package com.aiwellness.{server}.adapter.infrastructure.client.admin;

import com.aiwellness.{server}.domain.port.external.AdminServicePort;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Repository;

/**
 * Admin 서버 호출 Adapter
 * 외부 서비스(Feign Client) 호출을 담당하며, 헥사고날 구조 파악을 위한 로그를 포함합니다.
 */
@Slf4j
@Repository
@RequiredArgsConstructor
public class AdminServiceAdapter implements AdminServicePort {
    
    private final AdminServiceClient adminServiceClient;
    
    @Override
    public AdminResponse getAdmin(Long id) {
        log.info("[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 시작: service=admin-service, method=getAdmin, id={}", id);
        try {
            AdminResponse response = adminServiceClient.getAdmin(id);
            log.info("[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 완료: service=admin-service, method=getAdmin, id={}, success=true", id);
            return response;
        } catch (Exception e) {
            log.error("[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 실패: service=admin-service, method=getAdmin, id={}, error={}", id, e.getMessage());
            throw e;
        }
    }
    
    @Override
    public AdminResponse createAdmin(AdminCreateRequest request) {
        log.info("[Adapter/Infrastructure/Client] AdminServiceAdapter.createAdmin() - 외부 서비스 호출 시작: service=admin-service, method=createAdmin, email={}", request.getEmail());
        try {
            AdminResponse response = adminServiceClient.createAdmin(request);
            log.info("[Adapter/Infrastructure/Client] AdminServiceAdapter.createAdmin() - 외부 서비스 호출 완료: service=admin-service, method=createAdmin, id={}, email={}, success=true", 
                    response.getId(), request.getEmail());
            return response;
        } catch (Exception e) {
            log.error("[Adapter/Infrastructure/Client] AdminServiceAdapter.createAdmin() - 외부 서비스 호출 실패: service=admin-service, method=createAdmin, email={}, error={}", 
                    request.getEmail(), e.getMessage());
            throw e;
        }
    }
}
```

### 3.5. 에러 처리

Feign Client는 자동으로 에러를 처리합니다:

- **4xx 에러**: `FeignException` 발생
- **5xx 에러**: `FeignException` 발생

```java
try {
    AdminResponse admin = adminServicePort.getAdmin(id);
} catch (FeignException e) {
    if (e.status() == 404) {
        throw new AdminNotFoundException();
    } else if (e.status() >= 500) {
        throw new ServiceUnavailableException();
    }
}
```

## 5. SSO 연계 외부 API 호출

### 5.1. SSO 연계 개요

SSO(Single Sign-On) 연계가 필요한 외부 모듈 솔루션과 통신할 때는 다음과 같은 구조를 사용합니다:

```
┌─────────────────────────────────────┐
│   Application (Service)             │
│   - 비즈니스 로직 처리                │
└─────────────────────────────────────┘
              ↓ 사용
┌─────────────────────────────────────┐
│   Domain (Port)                     │
│   - ExternalSsoApiPort              │  ← 아웃바운드 포트
└─────────────────────────────────────┘
              ↓ 구현
┌─────────────────────────────────────┐
│   Infrastructure (Adapter)         │
│   - ExternalSsoApiAdapter           │
│     - SsoTokenManager              │  ← SSO 토큰 관리
│     - ExternalSsoApiClient         │  ← Feign Client
└─────────────────────────────────────┘
              ↓ 호출
┌─────────────────────────────────────┐
│   외부 SSO 솔루션 API                 │
└─────────────────────────────────────┘
```

### 5.2. 폴더 구조

```
{server}-server/src/main/java/com/aiwellness/{server}/
├── domain/
│   └── port/
│       └── external/                   # 외부 시스템 Port
│           └── ExternalSsoApiPort.java
│
└── adapter/
    └── infrastructure/
        └── external/                   # 외부 시스템 Adapter
            └── sso/                    # SSO 연계 외부 API
                ├── ExternalSsoApiAdapter.java
                ├── ExternalSsoApiClient.java
                ├── SsoTokenManager.java
                └── dto/
                    ├── SsoTokenRequest.java
                    ├── SsoTokenResponse.java
                    └── ExternalApiRequest.java
```

### 5.3. 구현 방법

#### 1단계: Common 모듈의 SSO 토큰 관리자 사용

Common 모듈에 `SsoTokenManager`가 제공되므로, 각 서버에서는 이를 주입받아 사용합니다:

```java
/**
 * Common 모듈의 SsoTokenManager를 사용합니다.
 * 각 서버에서는 이 유틸리티를 주입받아 사용하면 됩니다.
 */
```

#### 2단계: Domain Port 정의

#### 2단계: Feign Client 설정 (SSO 토큰 자동 주입)

각 서버에서 `SsoTokenManager`를 주입받아 사용합니다. **상속이 필요 없고, `super.getAccessToken()`을 사용할 필요도 없습니다.**

```java
package com.aiwellness.{server}.adapter.infrastructure.external.sso;

import com.aiwellness.common.util.sso.SsoTokenManager;
import feign.RequestInterceptor;
import feign.RequestTemplate;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * 외부 SSO API 호출을 위한 Feign Client 설정
 * SSO 토큰을 자동으로 주입합니다.
 * 
 * 각 서버(admin, manager, customer)에서 동일한 방식으로 사용합니다.
 */
@Slf4j
@Configuration
@RequiredArgsConstructor
public class ExternalSsoApiFeignConfig {
    
    // Common 모듈의 SsoTokenManager를 주입받아 사용
    // 상속이 필요 없고, super.getAccessToken()을 사용할 필요도 없음
    private final SsoTokenManager ssoTokenManager;
    
    @Bean
    public RequestInterceptor ssoTokenInterceptor() {
        return new RequestInterceptor() {
            @Override
            public void apply(RequestTemplate template) {
                // SsoTokenManager를 직접 사용 (상속 불필요)
                String accessToken = ssoTokenManager.getAccessToken("external-api");
                
                // Authorization 헤더에 Bearer 토큰 추가
                template.header("Authorization", "Bearer " + accessToken);
                
                log.debug("[ExternalSsoApiFeignConfig] SSO token injected for request: {}", 
                    template.url());
            }
        };
    }
}
```

**중요:**
- `SsoTokenManager`는 `@Component`로 등록되어 있어 각 서버에서 자동으로 주입 가능
- 상속(`extends SsoTokenManager`)이 필요 없음
- `super.getAccessToken()`을 사용할 필요 없음
- 그냥 `ssoTokenManager.getAccessToken(scope)`를 직접 호출하면 됨

#### 3단계: Adapter 구현

```java
package com.aiwellness.{server}.adapter.infrastructure.external.sso;

import com.aiwellness.common.util.sso.SsoTokenManager;

import com.aiwellness.{server}.domain.port.external.ExternalSsoApiPort;
import com.aiwellness.{server}.adapter.infrastructure.external.sso.dto.ExternalApiRequest;
import com.aiwellness.{server}.adapter.infrastructure.external.sso.dto.ExternalApiResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Repository;

/**
 * 외부 SSO 연계 API 호출 Adapter
 */
@Slf4j
@Repository
@RequiredArgsConstructor
public class ExternalSsoApiAdapter implements ExternalSsoApiPort {
    
    private final ExternalSsoApiClient externalSsoApiClient;
    
    @Override
    public ExternalApiResponse callExternalApi(ExternalApiRequest request) {
        log.info("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
            "외부 SSO API 호출 시작: service=external-sso-api, method=processRequest, " +
            "requestId={}", request.getRequestId());
        
        try {
            // SSO 토큰은 Feign Client 설정(ExternalSsoApiFeignConfig)에서 자동으로 주입됨
            // Adapter에서는 직접 토큰을 전달할 필요 없음
            ExternalApiResponse response = externalSsoApiClient.processRequest(request);
            
            log.info("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
                "외부 SSO API 호출 완료: service=external-sso-api, method=processRequest, " +
                "requestId={}, success=true", request.getRequestId());
            
            return response;
            
        } catch (Exception e) {
            log.error("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
                "외부 SSO API 호출 실패: service=external-sso-api, method=processRequest, " +
                "requestId={}, error={}", request.getRequestId(), e.getMessage(), e);
            throw new RuntimeException("Failed to call external SSO API: " + e.getMessage(), e);
        }
    }
}
```

### 5.8. Redis를 사용한 토큰 캐싱 (운영 환경 권장)

운영 환경에서는 Redis를 사용한 분산 캐싱을 권장합니다. 

**방법 1: 컴포지션 사용 (권장)**

`SsoTokenManager`를 주입받아 사용하는 방식입니다. 이 방법이 더 유연하고 테스트하기 쉽습니다:

```java
@Component
@RequiredArgsConstructor
@Slf4j
public class RedisSsoTokenManager {
    
    private final SsoTokenManager ssoTokenManager;  // Common 모듈의 SsoTokenManager 주입
    private final RedisTemplate<String, String> redisTemplate;
    private static final String TOKEN_CACHE_KEY = "sso:token:";
    
    public String getAccessToken(String scope) {
        String cacheKey = TOKEN_CACHE_KEY + (scope != null ? scope : "default");
        String cachedToken = redisTemplate.opsForValue().get(cacheKey);
        
        if (cachedToken != null) {
            log.debug("[RedisSsoTokenManager] Using cached token from Redis for scope: {}", scope);
            return cachedToken;
        }
        
        // Common 모듈의 SsoTokenManager를 사용하여 토큰 발급
        // 이 경우 로컬 캐시는 사용하지 않고 Redis만 사용
        SsoTokenResponse tokenResponse = ssoTokenManager.requestToken(scope);
        String token = tokenResponse.getAccessToken();
        
        // Redis에 토큰 저장 (TTL: 만료 시간 - 5분)
        long ttl = tokenResponse.getExpiresIn() - 300;
        redisTemplate.opsForValue().set(cacheKey, token, Duration.ofSeconds(ttl));
        
        return token;
    }
}
```

**방법 2: 상속 사용 (선택사항)**

상속을 사용하는 경우, `SsoTokenManager`의 `requestToken()` 메서드가 `protected`로 선언되어 있어 사용 가능합니다:

```java
@Component
@RequiredArgsConstructor
@Slf4j
public class RedisSsoTokenManager extends SsoTokenManager {
    
    private final RedisTemplate<String, String> redisTemplate;
    private static final String TOKEN_CACHE_KEY = "sso:token:";
    
    @Override
    public String getAccessToken(String scope) {
        String cacheKey = TOKEN_CACHE_KEY + (scope != null ? scope : "default");
        String cachedToken = redisTemplate.opsForValue().get(cacheKey);
        
        if (cachedToken != null) {
            log.debug("[RedisSsoTokenManager] Using cached token from Redis for scope: {}", scope);
            return cachedToken;
        }
        
        // 부모 클래스의 requestToken() 메서드 사용 (protected)
        SsoTokenResponse tokenResponse = requestToken(scope);
        String token = tokenResponse.getAccessToken();
        
        // Redis에 토큰 저장
        long ttl = tokenResponse.getExpiresIn() - 300;
        redisTemplate.opsForValue().set(cacheKey, token, Duration.ofSeconds(ttl));
        
        return token;
    }
}
```

**주의사항:**
- 상속을 사용하는 경우, `SsoTokenManager`의 로컬 캐시(`ConcurrentHashMap`)와 Redis 캐시가 중복될 수 있습니다
- 컴포지션 방식을 사용하면 로컬 캐시를 우회하고 Redis만 사용할 수 있어 더 명확합니다
- `super.getAccessToken(scope)`를 사용하면 부모 클래스의 로컬 캐시도 함께 사용되므로, Redis 캐싱과 중복될 수 있습니다

**권장: 방법 1 (컴포지션)**

외부 서비스 호출(Feign Client)은 **Adapter 레이어**에서 로그를 기록합니다:

**로그 형식:**

- 실패: `[Adapter/Infrastructure/Client] Adapter명.메서드명() - 외부 서비스 호출 실패: service=서비스명, method=메서드명, error=에러메시지`

**로그 흐름 예시:**
```
[Application/Service] OrderService.createOrder() - Use Case 시작: orderId=1
[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 시작: service=admin-service, method=getAdmin, id=1
[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 완료: service=admin-service, method=getAdmin, id=1, success=true
[Application/Service] OrderService.createOrder() - Use Case 완료: orderId=1
```

**주의사항:**
- Feign Client 인터페이스는 로그를 직접 찍을 수 없으므로, **Adapter에서 로그를 기록**합니다
- 외부 API 호출(HTTP Client, RestTemplate 등)도 동일한 형식으로 로그를 기록합니다
- 외부 시스템 연동의 경우 `[Adapter/Infrastructure/External]` 형식을 사용할 수 있습니다

#### 2단계: Feign Client 설정 (SSO 토큰 자동 주입)

각 서버에서 `SsoTokenManager`를 주입받아 사용합니다. **상속이 필요 없고, `super.getAccessToken()`을 사용할 필요도 없습니다.**

```java
package com.aiwellness.{server}.adapter.infrastructure.external.sso;

import com.aiwellness.common.util.sso.SsoTokenManager;
import feign.RequestInterceptor;
import feign.RequestTemplate;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * 외부 SSO API 호출을 위한 Feign Client 설정
 * SSO 토큰을 자동으로 주입합니다.
 * 
 * 각 서버(admin, manager, customer)에서 동일한 방식으로 사용합니다.
 */
@Slf4j
@Configuration
@RequiredArgsConstructor
public class ExternalSsoApiFeignConfig {
    
    // Common 모듈의 SsoTokenManager를 주입받아 사용
    // 상속이 필요 없고, super.getAccessToken()을 사용할 필요도 없음
    private final SsoTokenManager ssoTokenManager;
    
    @Bean
    public RequestInterceptor ssoTokenInterceptor() {
        return new RequestInterceptor() {
            @Override
            public void apply(RequestTemplate template) {
                // SsoTokenManager를 직접 사용 (상속 불필요)
                String accessToken = ssoTokenManager.getAccessToken("external-api");
                
                // Authorization 헤더에 Bearer 토큰 추가
                template.header("Authorization", "Bearer " + accessToken);
                
                log.debug("[ExternalSsoApiFeignConfig] SSO token injected for request: {}", 
                    template.url());
            }
        };
    }
}
```

**중요:**
- `SsoTokenManager`는 `@Component`로 등록되어 있어 각 서버에서 자동으로 주입 가능
- 상속(`extends SsoTokenManager`)이 필요 없음
- `super.getAccessToken()`을 사용할 필요 없음
- 그냥 `ssoTokenManager.getAccessToken(scope)`를 직접 호출하면 됨

#### 3단계: Adapter 구현

```java
package com.aiwellness.{server}.adapter.infrastructure.external.sso;

import com.aiwellness.common.util.sso.SsoTokenManager;

import com.aiwellness.{server}.domain.port.external.ExternalSsoApiPort;
import com.aiwellness.{server}.adapter.infrastructure.external.sso.dto.ExternalApiRequest;
import com.aiwellness.{server}.adapter.infrastructure.external.sso.dto.ExternalApiResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Repository;

/**
 * 외부 SSO 연계 API 호출 Adapter
 */
@Slf4j
@Repository
@RequiredArgsConstructor
public class ExternalSsoApiAdapter implements ExternalSsoApiPort {
    
    private final ExternalSsoApiClient externalSsoApiClient;
    
    @Override
    public ExternalApiResponse callExternalApi(ExternalApiRequest request) {
        log.info("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
            "외부 SSO API 호출 시작: service=external-sso-api, method=processRequest, " +
            "requestId={}", request.getRequestId());
        
        try {
            // SSO 토큰은 Feign Client 설정(ExternalSsoApiFeignConfig)에서 자동으로 주입됨
            // Adapter에서는 직접 토큰을 전달할 필요 없음
            ExternalApiResponse response = externalSsoApiClient.processRequest(request);
            
            log.info("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
                "외부 SSO API 호출 완료: service=external-sso-api, method=processRequest, " +
                "requestId={}, success=true", request.getRequestId());
            
            return response;
            
        } catch (Exception e) {
            log.error("[Adapter/Infrastructure/External] ExternalSsoApiAdapter.callExternalApi() - " +
                "외부 SSO API 호출 실패: service=external-sso-api, method=processRequest, " +
                "requestId={}, error={}", request.getRequestId(), e.getMessage(), e);
            throw new RuntimeException("Failed to call external SSO API: " + e.getMessage(), e);
        }
    }
}
```

### 5.5. 주의사항

#### 5.5.1. 토큰 캐싱

- **로컬 캐시**: 개발 환경에서는 `ConcurrentHashMap` 사용 가능
- **분산 캐시**: 운영 환경에서는 **Redis** 등 분산 캐시 사용 권장
- **토큰 만료**: 토큰 만료 시간 전에 자동 갱신하도록 구현

#### 5.5.2. 보안

- **클라이언트 시크릿**: 환경 변수로 관리, 절대 코드에 하드코딩 금지
- **토큰 전송**: HTTPS만 사용
- **토큰 저장**: 메모리에만 저장, 로그에 출력 금지

#### 5.5.3. 에러 처리

```java
try {
    ExternalApiResponse response = externalSsoApiPort.callExternalApi(request);
} catch (FeignException.Unauthorized e) {
    // 401 에러: 토큰 만료 또는 잘못된 토큰
    // 토큰 캐시 무효화 후 재시도
    ssoTokenManager.invalidateToken(scope);
    response = externalSsoApiPort.callExternalApi(request);
} catch (FeignException e) {
    // 기타 에러 처리
    log.error("External SSO API error: {}", e.getMessage());
    throw new ExternalApiException("Failed to call external API", e);
}
```

#### 5.5.4. 재시도 로직

```java
@Retryable(
    value = {FeignException.class},
    maxAttempts = 3,
    backoff = @Backoff(delay = 1000, multiplier = 2)
)
public ExternalApiResponse callExternalApi(ExternalApiRequest request) {
    // 재시도 로직이 자동으로 적용됨
    return externalSsoApiClient.processRequest(token, request);
}
```

### 5.8. Redis를 사용한 토큰 캐싱 (운영 환경 권장)

운영 환경에서는 Redis를 사용한 분산 캐싱을 권장합니다. 

**방법 1: 컴포지션 사용 (권장)**

`SsoTokenManager`를 주입받아 사용하는 방식입니다. 이 방법이 더 유연하고 테스트하기 쉽습니다:

```java
@Component
@RequiredArgsConstructor
@Slf4j
public class RedisSsoTokenManager {
    
    private final SsoTokenManager ssoTokenManager;  // Common 모듈의 SsoTokenManager 주입
    private final RedisTemplate<String, String> redisTemplate;
    private static final String TOKEN_CACHE_KEY = "sso:token:";
    
    public String getAccessToken(String scope) {
        String cacheKey = TOKEN_CACHE_KEY + (scope != null ? scope : "default");
        String cachedToken = redisTemplate.opsForValue().get(cacheKey);
        
        if (cachedToken != null) {
            log.debug("[RedisSsoTokenManager] Using cached token from Redis for scope: {}", scope);
            return cachedToken;
        }
        
        // Common 모듈의 SsoTokenManager를 사용하여 토큰 발급
        // 이 경우 로컬 캐시는 사용하지 않고 Redis만 사용
        SsoTokenResponse tokenResponse = ssoTokenManager.requestToken(scope);
        String token = tokenResponse.getAccessToken();
        
        // Redis에 토큰 저장 (TTL: 만료 시간 - 5분)
        long ttl = tokenResponse.getExpiresIn() - 300;
        redisTemplate.opsForValue().set(cacheKey, token, Duration.ofSeconds(ttl));
        
        return token;
    }
}
```

**방법 2: 상속 사용 (선택사항)**

상속을 사용하는 경우, `SsoTokenManager`의 `requestToken()` 메서드가 `protected`로 선언되어 있어 사용 가능합니다:

```java
@Component
@RequiredArgsConstructor
@Slf4j
public class RedisSsoTokenManager extends SsoTokenManager {
    
    private final RedisTemplate<String, String> redisTemplate;
    private static final String TOKEN_CACHE_KEY = "sso:token:";
    
    @Override
    public String getAccessToken(String scope) {
        String cacheKey = TOKEN_CACHE_KEY + (scope != null ? scope : "default");
        String cachedToken = redisTemplate.opsForValue().get(cacheKey);
        
        if (cachedToken != null) {
            log.debug("[RedisSsoTokenManager] Using cached token from Redis for scope: {}", scope);
            return cachedToken;
        }
        
        // 부모 클래스의 requestToken() 메서드 사용 (protected)
        SsoTokenResponse tokenResponse = requestToken(scope);
        String token = tokenResponse.getAccessToken();
        
        // Redis에 토큰 저장
        long ttl = tokenResponse.getExpiresIn() - 300;
        redisTemplate.opsForValue().set(cacheKey, token, Duration.ofSeconds(ttl));
        
        return token;
    }
}
```

**주의사항:**
- 상속을 사용하는 경우, `SsoTokenManager`의 로컬 캐시(`ConcurrentHashMap`)와 Redis 캐시가 중복될 수 있습니다
- 컴포지션 방식을 사용하면 로컬 캐시를 우회하고 Redis만 사용할 수 있어 더 명확합니다
- `super.getAccessToken(scope)`를 사용하면 부모 클래스의 로컬 캐시도 함께 사용되므로, Redis 캐싱과 중복될 수 있습니다

**권장: 방법 1 (컴포지션)**

#### Feign Client 로깅 설정

Feign Client 로깅은 `FeignConfig`에서 설정됩니다:

```java
@Bean
public Logger.Level feignLoggerLevel() {
    return Logger.Level.FULL;  // 요청/응답 전체 로깅
```

#### 헥사고날 구조 파악을 위한 외부 호출 로그

외부 서비스 호출(Feign Client)은 **Adapter 레이어**에서 로그를 기록합니다:

**로그 형식:**
- 시작: `[Adapter/Infrastructure/Client] Adapter명.메서드명() - 외부 서비스 호출 시작: service=서비스명, method=메서드명, 파라미터`
- 완료: `[Adapter/Infrastructure/Client] Adapter명.메서드명() - 외부 서비스 호출 완료: service=서비스명, method=메서드명, 결과`
- 실패: `[Adapter/Infrastructure/Client] Adapter명.메서드명() - 외부 서비스 호출 실패: service=서비스명, method=메서드명, error=에러메시지`

**로그 흐름 예시:**
```
[Application/Service] OrderService.createOrder() - Use Case 시작: orderId=1
[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 시작: service=admin-service, method=getAdmin, id=1
[Adapter/Infrastructure/Client] AdminServiceAdapter.getAdmin() - 외부 서비스 호출 완료: service=admin-service, method=getAdmin, id=1, success=true
[Application/Service] OrderService.createOrder() - Use Case 완료: orderId=1
```

**주의사항:**
- Feign Client 인터페이스는 로그를 직접 찍을 수 없으므로, **Adapter에서 로그를 기록**합니다
- 외부 API 호출(HTTP Client, RestTemplate 등)도 동일한 형식으로 로그를 기록합니다
- 외부 시스템 연동의 경우 `[Adapter/Infrastructure/External]` 형식을 사용할 수 있습니다
